+def test_lift_coverage_pads_and_edge_is_strip_boundary():
+    """T9 85 钉(两类根因): ①lift 覆盖含外弧角点越出段(pad = sin·(ring_t+

+    assert cov[1] == pytest.approx(1.0 + (ext - 0.004), abs=1e-12)
+    assert P._ring_lift_coverage({"stations": [-1.0, 1.0],
